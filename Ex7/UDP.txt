#include <stdlib.h>

#define BUFFER_SIZE 1024
#define USERS_FILE "users.txt"
#define MAX_USERS 100

typedef struct {
    char username[64];
    char password[64];
} User;

User users[MAX_USERS];
int user_count = 0;

void load_users() {
    FILE *fp = fopen(USERS_FILE, "r");
    if (!fp) {
        perror("Failed to open users.txt");
        exit(EXIT_FAILURE);
    }
    while (fscanf(fp, "%63s %63s", users[user_count].username, users[user_count].password) == 2) {
        user_count++;
        if (user_count >= MAX_USERS) break;
    }
    fclose(fp);
}

int authenticate(const char *username, const char *password) {
    for (int i = 0; i < user_count; i++) {
        if (strcmp(users[i].username, username) == 0 &&
            strcmp(users[i].password, password) == 0) {
            return 1;
        }
    }
    return 0;
}

void send_over(int sockfd, struct sockaddr_in *client_addr, socklen_t addr_len, const char *msg) {
    char buffer[BUFFER_SIZE];
    snprintf(buffer, BUFFER_SIZE, "%s OVER", msg);
    sendto(sockfd, buffer, strlen(buffer), 0,
           (struct sockaddr *)client_addr, addr_len);
    printf("[TX] %s OVER\n", msg);
}

int recv_over(int sockfd, char *buffer, struct sockaddr_in *client_addr, socklen_t *addr_len) {
    memset(buffer, 0, BUFFER_SIZE);
    int n = recvfrom(sockfd, buffer, BUFFER_SIZE, 0,
                     (struct sockaddr *)client_addr, addr_len);
    if (n < 0) {
        perror("Receive failed");
        return -1;
    }
    buffer[n] = '\0';

    char *suffix = " OVER";
    int len = strlen(buffer);
    int suffix_len = strlen(suffix);
    if (len >= suffix_len && strcmp(buffer + len - suffix_len, suffix) == 0) {
        buffer[len - suffix_len] = '\0';
    }

    printf("[RX] %s\n", buffer);
    return n;
}

void prompt_and_send(int sockfd, struct sockaddr_in *server_addr, socklen_t addr_len) {
    char input[BUFFER_SIZE];
    memset(input, 0, BUFFER_SIZE);
    if (fgets(input, BUFFER_SIZE, stdin) == NULL)
        return;
    input[strcspn(input, "\n")] = '\0';
    send_over(sockfd, server_addr, addr_len, input);
}

    int sockfd;
    struct sockaddr_in server_addr;
    socklen_t addr_len = sizeof(server_addr);

    while ((c = getchar()) != '\n' && c != EOF);

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(server_port);
    if (inet_pton(AF_INET, SERVER_IP, &server_addr.sin_addr) <= 0) {
        perror("Invalid address");
        close(sockfd);
        exit(EXIT_FAILURE);

    printf("Connecting to server at %s:%d\n\n", SERVER_IP, server_port);

    send_over(sockfd, &server_addr, addr_len, "CONNECT");

    int authenticated = 0;

            printf("\n%s\n", buffer);
            break;
        }

        printf("\n%s ", buffer);
        fflush(stdout);
        prompt_and_send(sockfd, &server_addr, addr_len);

            break;

            printf("\n%s\n", buffer);

        fflush(stdout);

            break;

            printf("\n%s\n", buffer);
            break;

            attempts++;

        close(sockfd);
